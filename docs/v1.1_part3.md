-- 同一构件同一时间只能有一个 active 主码
CREATE UNIQUE INDEX uq_qrcodes_component_active

- 主码：`status = 'active'` 且 `is_primary = TRUE`，同一构件唯一（部分索引保证）
- 历史码：`status IN ('disabled', 'voided')`，保留供追溯，不限制数量
